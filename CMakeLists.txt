cmake_minimum_required(VERSION 3.24)
project(MiyazatoSpatialEngine VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MIYAZATO_BUILD_DESKTOP "Build Windows WebView2 host" ${WIN32})
add_library(miyazato_dsp engine/dsp/Dsp.cpp)
target_include_directories(miyazato_dsp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
 target_compile_options(miyazato_dsp PRIVATE /W4 /permissive- /fp:precise)
else()
 target_compile_options(miyazato_dsp PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(FetchContent)
include(CTest)
if(BUILD_TESTING)
 FetchContent_Declare(Catch2 URL https://github.com/catchorg/Catch2/archive/refs/tags/v3.7.1.tar.gz URL_HASH SHA256=c991b247a1a0d7bb9c39aa35faf0fe9e19764213f28ffba3109388e62ee0269c DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
 FetchContent_MakeAvailable(Catch2)
 add_executable(dsp_tests tests/DspTests.cpp)
 target_link_libraries(dsp_tests PRIVATE miyazato_dsp Catch2::Catch2WithMain)
 add_test(NAME dsp_tests COMMAND dsp_tests)
 set_tests_properties(dsp_tests PROPERTIES TIMEOUT 30)
endif()
if(WIN32)
 add_library(miyazato_windows engine/audio/wasapi/Devices.cpp engine/audio/wasapi/LoopbackEngine.cpp)
 target_link_libraries(miyazato_windows PUBLIC miyazato_dsp ole32 uuid avrt)
 target_include_directories(miyazato_windows PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
 target_compile_definitions(miyazato_windows PUBLIC NOMINMAX UNICODE _UNICODE)
endif()
if(MIYAZATO_BUILD_DESKTOP)
 if(NOT WIN32)
  message(FATAL_ERROR "Desktop host requires Windows")
 endif()
 if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR CMAKE_SYSTEM_PROCESSOR MATCHES "ARM|arm")
  message(FATAL_ERROR "Desktop currently supports Windows x64 only")
 endif()
 set(WEBVIEW2_SDK_DIR "" CACHE PATH "Extracted Microsoft.Web.WebView2 NuGet package")
 if(NOT EXISTS "${WEBVIEW2_SDK_DIR}/build/native/include/WebView2.h")
  message(FATAL_ERROR "Set WEBVIEW2_SDK_DIR to the extracted WebView2 SDK")
 endif()
 FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
 FetchContent_MakeAvailable(json)
 enable_language(RC)
 configure_file("${CMAKE_CURRENT_SOURCE_DIR}/assets/branding/icons/miyazato-spatial-engine.ico"
  "${CMAKE_CURRENT_BINARY_DIR}/branding/miyazato-spatial-engine.ico" COPYONLY)
 add_executable(MiyazatoSpatialEngine WIN32 apps/desktop/native/main.cpp apps/desktop/native/app.rc)
 target_include_directories(MiyazatoSpatialEngine PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/apps/desktop/native")
 target_include_directories(MiyazatoSpatialEngine PRIVATE "${WEBVIEW2_SDK_DIR}/build/native/include")
 target_link_libraries(MiyazatoSpatialEngine PRIVATE miyazato_windows nlohmann_json::nlohmann_json "${WEBVIEW2_SDK_DIR}/build/native/x64/WebView2LoaderStatic.lib" user32 shell32 shlwapi version)
 target_compile_definitions(MiyazatoSpatialEngine PRIVATE NOMINMAX UNICODE _UNICODE)
 add_custom_target(miyazato_ui_assets ALL
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/apps/desktop/ui/dist" "$<TARGET_FILE_DIR:MiyazatoSpatialEngine>/ui"
  COMMENT "Copy bundled UI assets")
 add_dependencies(miyazato_ui_assets MiyazatoSpatialEngine)
 if(MSVC)
  target_compile_options(miyazato_windows PRIVATE /W4 /permissive-)
  target_compile_options(MiyazatoSpatialEngine PRIVATE /W4 /permissive-)
 endif()
endif()
